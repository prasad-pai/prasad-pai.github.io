const originalCount = books.length;
let sortKey = 'year';
let sortDir = -1;

const esc = s => String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const $ = id => document.getElementById(id);

function coverUrl(isbn) {
  return isbn ? `https://covers.openlibrary.org/b/isbn/${encodeURIComponent(isbn)}-M.jpg` : '';
}

function filtered() {
  const q = $('search').value.trim().toLowerCase();
  const y = $('year').value;
  const c = $('category').value;
  return books.filter(b =>
    (!q || (b.title + ' ' + b.author_display).toLowerCase().includes(q)) &&
    (!y || b.year === y) &&
    (!c || b.category === c)
  );
}

function renderStats() {
  $('ratedCount').textContent = books.filter(b => b.rating).length;
}

function renderYearCards() {
  const rows = {};
  filtered().forEach(b => rows[b.year] = (rows[b.year] || 0) + 1);
  $('yearCards').innerHTML = Object.entries(rows)
    .sort((a,b) => b[0].localeCompare(a[0]))
    .map(([y,n]) => `<div class="year-card"><strong>${n}</strong><span>${esc(y)} books</span></div>`)
    .join('') || '<p class="small">No matching books.</p>';
}

function renderAuthors() {
  const counts = {};
  filtered().forEach(b => { if (b.author_display) counts[b.author_display] = (counts[b.author_display]||0)+1; });
  $('authorList').innerHTML = Object.entries(counts)
    .sort((a,b) => b[1]-a[1] || a[0].localeCompare(b[0]))
    .slice(0,12)
    .map(([a,n]) => `<li><strong>${esc(a)}</strong> <span class="small">(${n})</span></li>`)
    .join('') || '<li>No matching authors.</li>';
}

function renderBars() {
  const rows = {};
  filtered().forEach(b => rows[b.category] = (rows[b.category]||0)+1);
  const max = Math.max(1, ...Object.values(rows));
  $('categoryBars').innerHTML = Object.entries(rows)
    .sort((a,b)=>b[1]-a[1])
    .map(([k,v]) => `<div class="bar-row"><span>${esc(k)}</span><div class="bar"><span style="width:${(v/max)*100}%"></span></div><strong>${v}</strong></div>`)
    .join('') || '<p class="small">No matching books.</p>';
}

function stars(b) {
  return [1,2,3,4,5].map(n =>
    `<button class="${Number(b.rating) >= n ? 'active' : ''}" title="${n}/5" data-rate="${n}">★</button>`
  ).join('');
}

function renderTable() {
  const rows = filtered().slice().sort((a,b) => {
    if (sortKey === 'favorite') return (Number(b.favorite)-Number(a.favorite))*sortDir;
    if (sortKey === 'rating') return (Number(a.rating||0)-Number(b.rating||0))*sortDir;
    const av = String(a[sortKey] ?? '').toLowerCase();
    const bv = String(b[sortKey] ?? '').toLowerCase();
    return av < bv ? -sortDir : av > bv ? sortDir : 0;
  });

  $('booksTable').querySelector('tbody').innerHTML = rows.map((b, idx) => {
    const cover = coverUrl(b.isbn);
    return `<tr data-index="${books.indexOf(b)}">
      <td><button class="fav" data-fav aria-label="Favorite">${b.favorite ? '♥' : '♡'}</button></td>
      <td>${cover ? `<img class="cover" src="${cover}" alt="Book cover">` : '<div class="cover-placeholder">📖</div>'}</td>
      <td><strong>${esc(b.title)}</strong>${b.notes ? `<div class="small">${esc(b.notes)}</div>` : ''}</td>
      <td>${esc(b.author_display || '—')}</td>
      <td>${esc(b.year)}</td>
      <td><span class="badge">${esc(b.category)}</span></td>
      <td class="stars">${stars(b)}</td>
    </tr>`;
  }).join('') || '<tr><td colspan="7">No books match the current filters.</td></tr>';

  $('resultsMeta').textContent = `Showing ${rows.length} of ${originalCount} books`;
}

function render() {
  renderStats();
  renderYearCards();
  renderAuthors();
  renderBars();
  renderTable();
}

['search','year','category'].forEach(id => $(id).addEventListener('input', render));

$('reset').addEventListener('click', () => {
  $('search').value=''; $('year').value=''; $('category').value=''; render();
});

document.querySelectorAll('#booksTable th').forEach(th => th.addEventListener('click', () => {
  const key = th.dataset.key;
  if (sortKey === key) sortDir *= -1;
  else { sortKey = key; sortDir = 1; }
  render();
}));

document.querySelector('#booksTable tbody').addEventListener('click', e => {
  const row = e.target.closest('tr[data-index]');
  if (!row) return;
  const b = books[Number(row.dataset.index)];
  if (e.target.closest('[data-fav]')) {
    b.favorite = !b.favorite;
    render();
  }
  const rate = e.target.closest('[data-rate]');
  if (rate) {
    const n = Number(rate.dataset.rate);
    b.rating = Number(b.rating) === n ? '' : n;
    render();
  }
});

render();
</script>
